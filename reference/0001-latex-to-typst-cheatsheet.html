Command line

# → file.pdf
typst compile file.typ
# → png (--ppi 144 for sharper)
typst compile file.typ out.png
# recompile on every save
typst watch file.typ
# list fonts Typst can see
typst fonts
# try an expression
typst eval '1 + 2'
